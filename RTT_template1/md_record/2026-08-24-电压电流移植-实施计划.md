### 2.2 `hc32_drv_adc.c` 移植规则（对照 HB Adc.c 逐函数）

- （阶段一不启用硬件触发；TMR0/AOS 见 Task 9）
- `Adc_IrqConfig`：`INTC_IrqSignIn(INT_SRC_ADC1_EOCA → INT116_IRQn)` + NVIC 配置 + `ADC_IntCmd(EOCA, ENABLE)`
- `ADC1_SeqA_IrqCallback` 主流程

**新增（EOCA 中断处理 + 软件触发）**：
```c
/* 软件触发：阶段一由 Dev_Adc_Task 周期调用 */
void AdcDrv_SoftwareTrigger(void) { ADC_Start(ADC_UNIT); }

/* EOCA 中断：遍历所有中断通道，读原始值→换算→写环形缓冲 */
static void AdcDrv_ProcessChannels(void)
{
    for (uint8_t i = 0; i < s_u8InstNum; i++) {
        adc_drv_inst_t *pstc = &s_astcInsts[i];
        uint16_t u16Raw = ADC_GetValue(ADC_UNIT, pstc->stcCh.u8Channel);
        pstc->u16LatestRaw = u16Raw;
        pstc->u32SampleCount++;
        float fVal = (float)u16Raw * ADC_VREF / (float)(1U << ADC_RES) * pstc->stcCh.fGain + pstc->stcCh.fOffset;
        if (pstc->stcCh.bAbs && fVal < 0.0f) fVal = -fVal;   /* 电流通道取绝对值 */
        pstc->fLatest = fVal;
        pstc->afRing[pstc->u16WriteIdx] = fVal;
        pstc->u16WriteIdx = (pstc->u16WriteIdx + 1U) % ADC_DRV_RING_SIZE;
        if (pstc->u16Count < ADC_DRV_RING_SIZE) pstc->u16Count++;
        if (pstc->stcCh.pfnCallback) pstc->stcCh.pfnCallback(u16Raw, pstc->stcCh.u8Channel);
    }
}
```

```c
#endif
```

### `dev_registry.c` 要点
```c
#define MAX_REG_MODULES 16
static SysModule_t s_modules[MAX_REG_MODULES];
static uint16_t    s_module_num;

int Dev_Registry_Add(const SysModule_t *module)
{
    if (!module || s_module_num >= MAX_REG_MODULES) return -1;
    for (uint16_t i = 0; i < s_module_num; i++)
        if (rt_strcmp(s_modules[i].name, module->name) == 0) return -1;
    s_modules[s_module_num++] = *module;
    if (module->init) module->init();   /* 注册即初始化（沿用源工程语义） */
    return 0;
}

void Dev_Registry_UpdateAll(void)
{
    uint32_t tick = rt_tick_get();
    for (uint16_t i = 0; i < s_module_num; i++) {
        SysModule_t *m = &s_modules[i];
        if (!m->enabled || !m->task) continue;
        if (m->period_ms == 0 || (tick - m->last_tick) >= m->period_ms) {
            m->task();
            m->last_tick = tick;
        }
    }
}

void Dev_Thread_Entry(void *param)
{
    (void)param;
    while (1) {
        Dev_Registry_UpdateAll();
        rt_thread_mdelay(1);
    }
}

int Dev_Start(void)
{
    rt_thread_t t = rt_thread_create("dev", Dev_Thread_Entry, RT_NULL,
                                     2048, 20, 10);
    if (t) rt_thread_startup(t);
    return t ? 0 : -1;
}
```

```c
/* current_sensor.c：读环形缓冲 → 均值 → 时间窗口过流 */
#include "current_sensor.h"
#include "dev_adc.h"
#include "dev_registry.h"

#define CUR_FILTER_N        (8U)        /* 滑动均值窗口 */
#define CUR_OVER_CUR_TH_MA  (10000.0f)  /* 过流阈值 10A，待整定 */
#define CUR_OVER_WINDOW_MS  (50U)       /* 判定时间窗口，待整定 */

static float    s_fCurrMa;
static uint8_t  s_u8Status;
static uint16_t s_u16OverCount;         /* 超阈值累计（折算为时间） */
static uint32_t s_u32LastTick;

void CurrentSensor_Init(void) { s_fCurrMa = 0; s_u8Status = 0; }

void CurrentSensor_Task(void)
{
    float fBuf[CUR_FILTER_N];
    uint16_t u16N = AdcDrv_ReadRing(1, fBuf, CUR_FILTER_N);
    float fSum = 0;
    for (uint16_t i = 0; i < u16N; i++) fSum += fBuf[i];
    if (u16N) s_fCurrMa = fSum / u16N;   /* 取环形缓冲前 N 个均值（近似） */

    /* 时间窗口过流：超阈值累计达到窗口时间才报 */
    uint32_t now = rt_tick_get();
    uint32_t dt = now - s_u32LastTick;
    s_u32LastTick = now;
    if (s_fCurrMa > CUR_OVER_CUR_TH_MA) {
        s_u16OverCount = (uint16_t)(s_u16OverCount + dt);
        if (s_u16OverCount >= CUR_OVER_WINDOW_MS) s_u8Status = 1;
    } else {
        s_u16OverCount = 0;
        s_u8Status = 0;
    }
}

void CurrentSensor_GetInfo(float *pfCurr_mA, uint8_t *pu8Status)
{
    if (pfCurr_mA)  *pfCurr_mA = s_fCurrMa;
    if (pu8Status) *pu8Status = s_u8Status;
}
```

```c
/* bus_voltage.c：读环形缓冲 → 均值 → 过压/欠压（迟滞 + 恢复延时） */
#include "bus_voltage.h"
#include "dev_adc.h"
#include "dev_registry.h"

#define VOL_FILTER_N        (20U)
#define VOL_OVER_TH         (32.0f)   /* 待整定 */
#define VOL_UNDER_TH        (8.0f)    /* 待整定 */
#define VOL_HYST            (2.0f)
#define VOL_RECOVER_DELAY_MS (3000U)

static float   s_fVolt;
static uint8_t s_u8Status;      /* 0 正常 1 欠压 2 过压 */
static uint8_t s_u8Fault;       /* 0 无 1 欠压 2 过压 */
static uint32_t s_u32RecoverStart;
static uint8_t s_u8Waiting;

void BusVoltage_Init(void) { s_fVolt = 0; s_u8Status = 0; s_u8Fault = 0; }

void BusVoltage_Task(void)
{
    float fBuf[VOL_FILTER_N];
    uint16_t u16N = AdcDrv_ReadRing(0, fBuf, VOL_FILTER_N);
    float fSum = 0;
    for (uint16_t i = 0; i < u16N; i++) fSum += fBuf[i];
    if (u16N) s_fVolt = fSum / u16N;

    uint32_t now = rt_tick_get();
    if (s_u8Fault != 0) {
        uint8_t recover = (s_u8Fault == 2) ? (s_fVolt < (VOL_OVER_TH - VOL_HYST))
                                           : (s_fVolt > (VOL_UNDER_TH + VOL_HYST));
        if (recover) {
            if (!s_u8Waiting) { s_u8Waiting = 1; s_u32RecoverStart = now; }
            else if ((now - s_u32RecoverStart) >= VOL_RECOVER_DELAY_MS) { s_u8Fault = 0; s_u8Waiting = 0; }
        } else { s_u8Waiting = 0; }
    } else {
        if (s_fVolt > VOL_OVER_TH)      s_u8Fault = 2;
        else if (s_fVolt < VOL_UNDER_TH) s_u8Fault = 1;
    }
```
