static void Motor\_ArbitrationDecision(MotorDevice\_t\* motor) {

&#x20;   if (!motor->enable) return;



&#x20;   // 1. 选出候选命?

&#x20;   MotorControlCommand\_t\* fwd\_cmd = NULL;

&#x20;   MotorControlCommand\_t\* rev\_cmd = NULL;



&#x20;   if(motor->block\_fwd.count == 0 \&\& motor->allow\_fwd.count > 0) {

&#x20;       fwd\_cmd = \&motor->allow\_fwd.commands\[0];

&#x20;   }



&#x20;   if(motor->block\_rev.count == 0 \&\& motor->allow\_rev.count > 0) {

&#x20;       rev\_cmd = \&motor->allow\_rev.commands\[0];

&#x20;   }



&#x20;   // 2. 冲突判断

&#x20;   MotorControlCommand\_t\* final = NULL;

&#x20;   motor->debug\_info.conflict\_fault = false;



&#x20;   if (fwd\_cmd \&\& rev\_cmd) {

&#x20;       if (fwd\_cmd->device\_id == rev\_cmd->device\_id) {

&#x20;           final = NULL;

&#x20;           motor->debug\_info.conflict\_fault = true;

&#x20;       } else if (fwd\_cmd->priority < rev\_cmd->priority) {

&#x20;           final = fwd\_cmd;

&#x20;       } else if (rev\_cmd->priority < fwd\_cmd->priority) {

&#x20;           final = rev\_cmd;

&#x20;       } else {

&#x20;           final = NULL;

&#x20;       }

&#x20;   } else {

&#x20;       final = fwd\_cmd ? fwd\_cmd : rev\_cmd;

&#x20;   }



&#x20;   // 3. 执行决策 - 带调试打?

&#x20;   if (final) {

&#x20;       MOTOR\_DEBUG("Arbitration: Selected %s (prio=%d), dir=%s, duty=%.1f%%\\r\\n",

&#x20;                   (final->device\_id == DEV\_ID\_IO\_FWD) ? "IO\_FWD" :

&#x20;                   (final->device\_id == DEV\_ID\_IO\_REV) ? "IO\_REV" :

&#x20;                   (final->device\_id == DEV\_ID\_RTURN\_FWD) ? "RTURN\_FWD" :

&#x20;                   (final->device\_id == DEV\_ID\_RTURN\_REV) ? "RTURN\_REV" :

&#x20;                   (final->device\_id == DEV\_ID\_OVERCUR\_FWD) ? "OVERCUR\_FWD" :

&#x20;                   (final->device\_id == DEV\_ID\_CAN) ? "CAN" :

&#x20;                   (final->device\_id == DEV\_ID\_POWER\_POS) ? "POWER\_POS" :

&#x20;                   (final->device\_id == DEV\_ID\_POWER\_NEG) ? "POWER\_NEG" : "UNKNOWN",

&#x20;                   final->priority,

&#x20;                   (final->type == CMD\_TYPE\_RUN\_FWD || final->type == CMD\_TYPE\_RAMP\_FWD) ? "FWD" : "REV",

&#x20;                   final->param);



&#x20;       motor->active\_device\_id = final->device\_id;

&#x20;       motor->active\_dir = (final->type == CMD\_TYPE\_RUN\_FWD || final->type == CMD\_TYPE\_RAMP\_FWD) ? DIR\_FWD : DIR\_REV;

&#x20;       motor->last\_sent\_duty = final->param;

&#x20;       motor->state = (final->type == CMD\_TYPE\_RAMP\_FWD || final->type == CMD\_TYPE\_RAMP\_REV) ? MS\_RAMPING : MS\_RUNNING;



&#x20;       // 调用仲裁回调

&#x20;       if (motor->active\_dir == DIR\_FWD) {

&#x20;           Motor\_OnArbitrationFwd(motor, motor->last\_sent\_duty);

&#x20;       } else {

&#x20;           Motor\_OnArbitrationRev(motor, motor->last\_sent\_duty);

&#x20;       }



&#x20;       // 更新PWM设备

&#x20;       DeviceCommandData\_t cmd\_data;

&#x20;       PWM\_RampCmd\_t ramp\_cmd;



&#x20;       if (final->type == CMD\_TYPE\_RAMP\_FWD || final->type == CMD\_TYPE\_RAMP\_REV) {

&#x20;           cmd\_data.cmd = CMD\_PWM\_SET\_DUTY\_SLOW;

&#x20;           ramp\_cmd.targetDuty = (uint16\_t)(final->param);

&#x20;           ramp\_cmd.rampTimeMs = 2000;  // 2秒缓?

&#x20;           cmd\_data.params = \&ramp\_cmd;

&#x20;           cmd\_data.param\_size = sizeof(PWM\_RampCmd\_t);

&#x20;       } else {

&#x20;           cmd\_data.cmd = CMD\_PWM\_SET\_DUTY;

&#x20;           cmd\_data.params = \&final->param;

&#x20;           cmd\_data.param\_size = sizeof(uint16\_t);

&#x20;       }

&#x20;       Device\_Control(DEV\_PWM\_MOTOR\_ID, \&cmd\_data);

&#x20;   } else {

&#x20;       MOTOR\_DEBUG("Arbitration: No command selected - STOP\\r\\n");



&#x20;       motor->last\_sent\_duty = 0.0f;

&#x20;       motor->state = MS\_IDLE;

&#x20;       motor->active\_dir = DIR\_NONE;

&#x20;       motor->active\_device\_id = DEV\_ID\_NONE;



&#x20;       // 调用仲裁停止回调

&#x20;       Motor\_OnArbitrationStop(motor);



&#x20;       // 停止PWM

&#x20;       DeviceCommandData\_t cmd\_data;

&#x20;       cmd\_data.cmd = CMD\_PWM\_STOP;

&#x20;       cmd\_data.params = NULL;

&#x20;       Device\_Control(DEV\_PWM\_MOTOR\_ID, \&cmd\_data);

&#x20;   }



&#x20;   Motor\_UpdateDebugInfo(motor);

}

&#x20;   DeviceNode\_t\* node = DeviceManager\_Get(DEV\_MOTOR\_ID);

&#x20;   if (!node || !node->private\_data) return;

&#x20;   if (ev->u8IsActive) {

&#x20;           dev\_id\_rev = DEV\_ID\_UNDERVOLTAGE\_REV;

&#x20;           alarm\_name = "UNDERVOLTAGE";

&#x20;       }



&#x20;       MotorControlCommand\_t block\_fwd\_cmd = {

&#x20;           .device\_id = dev\_id\_fwd,

&#x20;           .priority = PRIO\_LIMIT,

&#x20;           .type = CMD\_TYPE\_BLOCK\_FWD,

&#x20;           .timestamp = tickTimer\_GetCount()

&#x20;       MotorControlCommand\_t block\_rev\_cmd = {

&#x20;           .device\_id = dev\_id\_rev,

&#x20;           .priority = PRIO\_LIMIT,

&#x20;           .type = CMD\_TYPE\_BLOCK\_REV,

&#x20;           .timestamp = tickTimer\_GetCount()

&#x20;       };

&#x20;       Motor\_CmdList\_SetBlock(\&motor->block\_rev, block\_rev\_cmd);



&#x20;       MAIN\_D("Motor: %s alarm! Bus=%lu mV, Threshold=%lu mV - Block both directions!\\r\\n",

&#x20;              alarm\_name, ev->u32BusVoltageMv, ev->u32ThresholdMv);

&#x20;   } else {

&#x20;       // 报警解除 - 根据报警类型使用对应的设备ID? block\_fwd/block\_rev 移除阻塞指令

&#x20;       MotorDeviceId\_t dev\_id\_fwd, dev\_id\_rev;

&#x20;       const char\* alarm\_name;



&#x20;       if (ev->u8AlarmType == VOLTAGE\_ALARM\_OVERVOLTAGE) {

&#x20;           dev\_id\_fwd = DEV\_ID\_OVERVOLTAGE\_FWD;

&#x20;           dev\_id\_rev = DEV\_ID\_OVERVOLTAGE\_REV;

&#x20;           alarm\_name = "OVERVOLTAGE";

&#x20;       } else {

&#x20;           dev\_id\_fwd = DEV\_ID\_UNDERVOLTAGE\_FWD;

&#x20;           dev\_id\_rev = DEV\_ID\_UNDERVOLTAGE\_REV;

&#x20;           alarm\_name = "UNDERVOLTAGE";

&#x20;       }

&#x20;       Motor\_CmdList\_Remove(\&motor->block\_rev, dev\_id\_rev);



&#x20;       MAIN\_D("Motor: %s released! Bus=%lu mV - Unblock both directions\\r\\n",

&#x20;              alarm\_name, ev->u32BusVoltageMv);

&#x20;   }



&#x20;   Motor\_ArbitrationDecision(motor);

}



void Motor\_OnRTurnLimit(void\* payload) {

&#x20;   RTurn\_LimitEvent\_t\* ev = (RTurn\_LimitEvent\_t\*)payload;



&#x20;   DeviceNode\_t\* node = DeviceManager\_Get(DEV\_MOTOR\_ID);

&#x20;   if (!node || !node->private\_data) return;



&#x20;   MotorDevice\_t\* motor = (MotorDevice\_t\*)node->private\_data;



&#x20;   if (ev->u8IsActive) {

&#x20;       // 限位触发：插? BLOCK 指令（使? RTurn 专用设备ID?

&#x20;       MotorControlCommand\_t cmd = {

&#x20;           .device\_id = (ev->u8Direction == RTURN\_LIMIT\_FORWARD) ? DEV\_ID\_RTURN\_FWD : DEV\_ID\_RTURN\_REV,

&#x20;           .priority = PRIO\_LIMIT,

&#x20;           .timestamp = tickTimer\_GetCount()

&#x20;       };



&#x20;       if (ev->u8Direction == RTURN\_LIMIT\_FORWARD) {

&#x20;           cmd.type = CMD\_TYPE\_BLOCK\_FWD;

&#x20;           Motor\_CmdList\_SetBlock(\&motor->block\_fwd, cmd);

&#x20;           cmd.type = CMD\_TYPE\_BLOCK\_REV;

&#x20;           Motor\_CmdList\_SetBlock(\&motor->block\_rev, cmd);

&#x20;           MAIN\_D("Motor: RTurn reverse limit triggered, block reverse + clear allow\_rev\\r\\n");

&#x20;       }

&#x20;   } else {

&#x20;       // 限位解除：移? BLOCK 指令（使? RTurn 专用设备ID?

&#x20;       MotorDeviceId\_t id = (ev->u8Direction == RTURN\_LIMIT\_FORWARD) ? DEV\_ID\_RTURN\_FWD : DEV\_ID\_RTURN\_REV;

&#x20;       if (ev->u8Direction == RTURN\_LIMIT\_FORWARD) {

&#x20;           Motor\_CmdList\_Remove(\&motor->block\_fwd, id);

&#x20;       } else if (ev->u8Direction == RTURN\_LIMIT\_REVERSE) {

&#x20;           Motor\_CmdList\_Remove(\&motor->block\_rev, id);

&#x20;           MAIN\_D("Motor: RTurn reverse limit released\\r\\n");

&#x20;       } else {

&#x20;           // 方向?0时，移除两个方向

&#x20;           Motor\_CmdList\_Remove(\&motor->block\_fwd, DEV\_ID\_RTURN\_FWD);

&#x20;           Motor\_CmdList\_Remove(\&motor->block\_rev, DEV\_ID\_RTURN\_REV);

&#x20;       }

&#x20;   }



&#x20;   Motor\_ArbitrationDecision(motor);

}

&#x20;   DeviceNode\_t\* node = DeviceManager\_Get(DEV\_MOTOR\_ID);

&#x20;   if (!node || !node->private\_data) return;



&#x20;   MotorDevice\_t\* motor = (MotorDevice\_t\*)node->private\_data;



&#x20;   if (ev->u8IsActive) {

&#x20;               .device\_id = DEV\_ID\_OVERCUR\_FWD,

&#x20;           };

&#x20;       }

&#x20;       else

&#x20;       {

&#x20;           // 反转或停止时过流 -> 不阻塞，这是预期的到位堵?

void Motor\_OnManualIO(void\* payload) {

&#x20;   MotorManualIOEvent\_t\* ev = (MotorManualIOEvent\_t\*)payload;



&#x20;   // 先检查接收数据是否有?

&#x20;   if (ev == NULL) {

&#x20;       MAIN\_D("\[MOTOR] OnManualIO: NULL payload!\\r\\n");

&#x20;       return;

&#x20;   }



&#x20;   // 打印原始事件数据，使用十六进制打印便于分?

&#x20;   uint32\_t speed\_raw = \*(uint32\_t\*)\&ev->speed;

&#x20;   MAIN\_D("\[MOTOR] OnManualIO RAW: dir=%d, type=%d, speed\_raw=0x%08X\\r\\n",

&#x20;          ev->dir, ev->type, speed\_raw);



&#x20;   DeviceNode\_t\* node = DeviceManager\_Get(DEV\_MOTOR\_ID);

&#x20;   if (!node || !node->private\_data) {

&#x20;       MAIN\_D("\[MOTOR] OnManualIO: Device not found!\\r\\n");

&#x20;       return;

&#x20;   }



&#x20;   MotorDevice\_t\* motor = (MotorDevice\_t\*)node->private\_data;



&#x20;   // 根据方向确定是哪个IO设备 - 避免使用默认?

&#x20;   MotorDeviceId\_t io\_dev\_id = DEV\_ID\_NONE;  // 初始化为 NONE



&#x20;   if (ev->dir == DIR\_FWD) {

&#x20;       io\_dev\_id = DEV\_ID\_IO\_FWD;

&#x20;   } else if (ev->dir == DIR\_REV) {

&#x20;       io\_dev\_id = DEV\_ID\_IO\_REV;

&#x20;   } else if (ev->dir == DIR\_NONE) {

&#x20;       io\_dev\_id = DEV\_ID\_NONE;

&#x20;   } else {

&#x20;       // 无效? dir ?

&#x20;       MAIN\_D("\[MOTOR] OnManualIO: Invalid dir=%d, expected 0,1,2\\r\\n", ev->dir);

&#x20;       return;

&#x20;   }



&#x20;   const MotorDeviceGene\_t\* gene = NULL;

&#x20;   if (io\_dev\_id != DEV\_ID\_NONE) {

&#x20;       gene = Motor\_GetGene(io\_dev\_id);

&#x20;       if (!gene) {

&#x20;           MAIN\_D("\[MOTOR] OnManualIO: gene not found for dev\_id=%d!\\r\\n", io\_dev\_id);

&#x20;           return;

&#x20;       }

&#x20;   }



&#x20;   // 打印关键信息 - 将浮点数转换为整数打?

&#x20;   int32\_t speed\_int = (int32\_t)(ev->speed \* 10);

&#x20;   MAIN\_D("\[MOTOR] OnManualIO: dir=%d, type=%d, speed=%ld.%ld%%, io\_dev\_id=%d\\r\\n",

&#x20;          ev->dir, ev->type, (long)(speed\_int / 10), (long)(speed\_int % 10), io\_dev\_id);

&#x20;   MAIN\_D("\[MOTOR] OnManualIO: Before - allow\_fwd.count=%d, allow\_rev.count=%d, block\_fwd.count=%d, block\_rev.count=%d\\r\\n",

&#x20;          motor->allow\_fwd.count, motor->allow\_rev.count, motor->block\_fwd.count, motor->block\_rev.count);



&#x20;   // 处理具体指令

&#x20;   if (ev->type == CMD\_TYPE\_RUN\_FWD || ev->type == CMD\_TYPE\_RAMP\_FWD) {

&#x20;       // 检? io\_dev\_id 是否正确

&#x20;       if (io\_dev\_id != DEV\_ID\_IO\_FWD) {

&#x20;           MAIN\_D("\[MOTOR] ERROR: RUN\_FWD but io\_dev\_id=%d (expected %d)\\r\\n", io\_dev\_id, DEV\_ID\_IO\_FWD);

&#x20;           return;

&#x20;       }



&#x20;       // IO\_FWD运行：移除自己的BLOCK，添加自己的ALLOW

&#x20;       MotorControlCommand\_t cmd = {

&#x20;           .device\_id = io\_dev\_id,

&#x20;           .priority = gene->priority,

&#x20;           .type = ev->type,

&#x20;           .param = ev->speed,

&#x20;           .timestamp = tickTimer\_GetCount()

&#x20;       };



&#x20;       Motor\_CmdList\_Remove(\&motor->block\_fwd, io\_dev\_id);

&#x20;       Motor\_CmdList\_SetAllow(\&motor->allow\_fwd, \&motor->block\_fwd, cmd);



&#x20;       MAIN\_D("\[MOTOR] IO\_FWD: RUN, removed block\_fwd, added allow\_fwd\\r\\n");

&#x20;   }

&#x20;   else if (ev->type == CMD\_TYPE\_RUN\_REV || ev->type == CMD\_TYPE\_RAMP\_REV) {

&#x20;       // 检? io\_dev\_id 是否正确

&#x20;       if (io\_dev\_id != DEV\_ID\_IO\_REV) {

&#x20;           MAIN\_D("\[MOTOR] ERROR: RUN\_REV but io\_dev\_id=%d (expected %d)\\r\\n", io\_dev\_id, DEV\_ID\_IO\_REV);

&#x20;           return;

&#x20;       }



&#x20;       // IO\_REV运行：移除自己的BLOCK，添加自己的ALLOW

&#x20;       MotorControlCommand\_t cmd = {

&#x20;           .device\_id = io\_dev\_id,

&#x20;           .priority = gene->priority,

&#x20;           .type = ev->type,

&#x20;           .param = ev->speed,

&#x20;           .timestamp = tickTimer\_GetCount()

&#x20;       };



&#x20;       Motor\_CmdList\_Remove(\&motor->block\_rev, io\_dev\_id);

&#x20;       Motor\_CmdList\_SetAllow(\&motor->allow\_rev, \&motor->block\_rev, cmd);



&#x20;       MAIN\_D("\[MOTOR] IO\_REV: RUN, removed block\_rev, added allow\_rev\\r\\n");

&#x20;   }

&#x20;   else if (ev->type == CMD\_TYPE\_STOP) {

&#x20;       if (ev->dir == DIR\_FWD) {

&#x20;           // IO\_FWD停止：移除自己的ALLOW，添加自己的BLOCK

&#x20;           if (gene == NULL) {

&#x20;               gene = Motor\_GetGene(DEV\_ID\_IO\_FWD);

&#x20;               if (!gene) {

&#x20;                   MAIN\_D("\[MOTOR] OnManualIO: gene not found for IO\_FWD!\\r\\n");

&#x20;                   return;

&#x20;               }

&#x20;           }



&#x20;           Motor\_CmdList\_Remove(\&motor->allow\_fwd, DEV\_ID\_IO\_FWD);



&#x20;           MotorControlCommand\_t block\_cmd = {

&#x20;               .device\_id = DEV\_ID\_IO\_FWD,

&#x20;               .priority = gene->priority,

&#x20;               .type = CMD\_TYPE\_BLOCK\_FWD,

&#x20;               .timestamp = tickTimer\_GetCount()

&#x20;           };

&#x20;           // Motor\_CmdList\_SetBlock(\&motor->block\_fwd, block\_cmd);



&#x20;           MAIN\_D("\[MOTOR] IO\_FWD: STOP, removed allow\_fwd, added block\_fwd\\r\\n");

&#x20;       }

&#x20;       else if (ev->dir == DIR\_REV) {

&#x20;           // IO\_REV停止：移除自己的ALLOW，添加自己的BLOCK

&#x20;           if (gene == NULL) {

&#x20;               gene = Motor\_GetGene(DEV\_ID\_IO\_REV);

&#x20;               if (!gene) {

&#x20;                   MAIN\_D("\[MOTOR] OnManualIO: gene not found for IO\_REV!\\r\\n");

&#x20;                   return;

&#x20;               }

&#x20;           }



&#x20;           Motor\_CmdList\_Remove(\&motor->allow\_rev, DEV\_ID\_IO\_REV);



&#x20;           MotorControlCommand\_t block\_cmd = {

&#x20;               .device\_id = DEV\_ID\_IO\_REV,

&#x20;               .priority = gene->priority,

&#x20;               .type = CMD\_TYPE\_BLOCK\_REV,

&#x20;               .timestamp = tickTimer\_GetCount()

&#x20;           };

&#x20;           // Motor\_CmdList\_SetBlock(\&motor->block\_rev, block\_cmd);



&#x20;           MAIN\_D("\[MOTOR] IO\_REV: STOP, removed allow\_rev, added block\_rev\\r\\n");

&#x20;       }

&#x20;       else {

&#x20;           // 停止所有IO设备：移除各自的ALLOW，添加各自的BLOCK

&#x20;           const MotorDeviceGene\_t\* gene\_fwd = Motor\_GetGene(DEV\_ID\_IO\_FWD);

&#x20;           const MotorDeviceGene\_t\* gene\_rev = Motor\_GetGene(DEV\_ID\_IO\_REV);



&#x20;           Motor\_CmdList\_Remove(\&motor->allow\_fwd, DEV\_ID\_IO\_FWD);

&#x20;           Motor\_CmdList\_Remove(\&motor->allow\_rev, DEV\_ID\_IO\_REV);



&#x20;           if (gene\_fwd) {

&#x20;               MotorControlCommand\_t block\_fwd\_cmd = {

&#x20;                   .device\_id = DEV\_ID\_IO\_FWD,

&#x20;                   .priority = gene\_fwd->priority,

&#x20;                   .type = CMD\_TYPE\_BLOCK\_FWD,

&#x20;                   .timestamp = tickTimer\_GetCount()

&#x20;               };

&#x20;               // Motor\_CmdList\_SetBlock(\&motor->block\_fwd, block\_fwd\_cmd);

&#x20;           }



&#x20;           if (gene\_rev) {

&#x20;               MotorControlCommand\_t block\_rev\_cmd = {

&#x20;                   .device\_id = DEV\_ID\_IO\_REV,

&#x20;                   .priority = gene\_rev->priority,

&#x20;                   .type = CMD\_TYPE\_BLOCK\_REV,

&#x20;                   .timestamp = tickTimer\_GetCount()

&#x20;               };

&#x20;               // Motor\_CmdList\_SetBlock(\&motor->block\_rev, block\_rev\_cmd);

&#x20;           }



&#x20;           MAIN\_D("\[MOTOR] All IO: STOP, removed all allow, added all block\\r\\n");

&#x20;       }

&#x20;   }

&#x20;   else {

&#x20;       MAIN\_D("\[MOTOR] OnManualIO: Unknown type=%d\\r\\n", ev->type);

&#x20;       return;

&#x20;   }



&#x20;   Motor\_ArbitrationDecision(motor);

}
